- [ ] **Step 3: `test-runner.js` — `renderIntro(test)`**

`show("intro")` 전에 호출. meta로 채우고 디스커버/FAQ 토글:
```js
function renderIntro(test) {
  const m = test.meta;
  const set = (id, v) => { const el = document.getElementById(id); if (el) el.textContent = v; };
  set("intro-title", m.name);
  set("intro-lead", m.introLead || m.tagline || "");
  set("stat-count", String(test.questions.length));
  set("stat-duration", (m.durationMin ? m.durationMin + "분" : "—"));
  set("intro-sample", m.sampleQuestion || "");
  const startBtn = document.getElementById("intro-start"); if (startBtn) startBtn.textContent = `${m.name} 시작하기`;
  // 디스커버: renderDiscoverPreviews 있는 모듈만
  const disc = document.getElementById("intro-discover");
  if (disc) { if (test.renderDiscoverPreviews) { test.renderDiscoverPreviews(document); disc.hidden = false; } else disc.hidden = true; }
  // FAQ: meta.faq 있으면 렌더, 없으면 숨김
  const faqSec = document.getElementById("intro-faq"), faqList = document.getElementById("faq-list");
  if (faqSec && faqList) {
    if (m.faq && m.faq.length) {
      faqList.replaceChildren(...m.faq.map((f) => {
        const d = document.createElement("details");
        const s = document.createElement("summary"); s.textContent = f.q;
        const p = document.createElement("p"); p.textContent = f.a;
        d.append(s, p); return d;
      }));
      faqSec.hidden = false;
    } else faqSec.hidden = true;
  }
}
```
러너 초기화에서 모듈 해석 후 `renderIntro(test)` 호출(이전 `test.renderDiscoverPreviews?.(document)` 직접 호출은 renderIntro로 이동). 통계 라벨(문항/소요)·예시 스케일 점 등 정적 부분은 유지.

- [ ] **Step 1: `tests/egen-teto/questions.js` — 원작 20문항**

```js
const ET_AXES = { teto: ["에겐","테토"], extro: ["내향","외향"] };  // [0쪽, 100쪽]
const ET_SCALE = [ {value:1,label:"전혀 아니다"},{value:2,label:"아니다"},{value:3,label:"보통이다"},{value:4,label:"그렇다"},{value:5,label:"매우 그렇다"} ];
const ET_QUESTIONS = [ /* 총 20: teto 10 + extro 10, 각 축 +/− 균형 */ ];
```
각 문항 `{ text, axis:"teto"|"extro", keyed:"+"|"-" }`. `keyed:"+"` = 응답 클수록 해당 성향(테토/외향) 강함. teto+ 예: "갈등이 생기면 피하지 않고 정면으로 부딪힌다."; teto− 예: "결정 전에 상대의 감정을 먼저 살핀다."; extro+ 예: "처음 보는 자리에서도 먼저 말을 건다."; extro− 예: "혼자 정리하는 시간이 있어야 에너지가 찬다." (구현자: 자연스러운 원작 한국어 20문항, 축당 10·+/− 균형.)

- [ ] **Step 2: `tests/egen-teto/types.js` — 4유형**

const ET_TYPES = {
  TO: { title:"불꽃 리더", emoji:"🔥", summary:"…", description:"…", strengths:[…3], charm:"…매력 포인트…", attracted:"…끌리는 상대(에겐 계열)…" },

(구현자: 4유형 원작 — 테토=주도·직설·추진, 에겐=다정·섬세·공감; 외향=표현·활발, 내향=절제·정돈. `attracted`는 밈의 "서로 다른 기질에 끌림" 반영. summary 한 줄, description 1~2문단, strengths 3, charm·attracted 각 1~2문장.)

- [ ] **Step 3: `tests/egen-teto/module.js` — ET_MODULE**

```js
const ET_MODULE = {
  meta: { id:"egen-teto", name:"에겐·테토 기질 테스트", tagline:"주도냐 수용이냐 — 나의 기질 성향",
          icon:"⚡", scaleSize:5, scaleLabels: ET_SCALE.map(s=>s.label),
          introLead:"성별과 무관한 '기질 성향' 테스트. 20문항으로 테토(주도)–에겐(수용), 외향–내향 성향을 읽어요.",
          durationMin:3, sampleQuestion:"갈등이 생기면 피하지 않고 정면으로 부딪힌다." },
  questions: ET_QUESTIONS,
  score(answers) {   // { teto, extro } 각 0~100 (teto=테토 성향%, extro=외향 성향%)
    const sum={teto:0,extro:0}, cnt={teto:0,extro:0};
    ET_QUESTIONS.forEach((q,i)=>{ const v=answers[i]??3; const d=q.keyed==="+"?v:6-v; sum[q.axis]+=d; cnt[q.axis]+=1; });
    const s={}; ["teto","extro"].forEach(a=>{ const avg=cnt[a]?sum[a]/cnt[a]:3; s[a]=Math.round(((avg-1)/4)*100); });
    return s;
  },
  buildProfile(scores) {
    const code = (scores.teto>=50?"T":"E") + (scores.extro>=50?"O":"I");
    const t = ET_TYPES[code] || {};
    return { type:{code, title:t.title||code, role:t.title||"", emoji:t.emoji||"⚡"},
             axis:{teto:scores.teto, extro:scores.extro}, scores,
             summary:t.summary||"", description:t.description||"", strengths:t.strengths||[],
             charm:t.charm||"", attracted:t.attracted||"" };
  },
  renderResult(mountEl,{scores,profile}) { /* 유형명 대형+이모지, 스펙트럼 바 2개(에겐—테토, 내향—외향), 설명, 강점, 매력 포인트, 끌리는 상대 */ },
  renderSummaryViz(container,scores) { /* 스펙트럼 바 2개(컴팩트) */ },
  renderCompareViz(container,datasets) { /* 축별 두 값 나란히 바 */ },
};
```
바 시각화: 기존 `.dich`(dich-row: 좌 라벨 · 트랙+마커 · 우 라벨) 재사용. teto행 = 좌 "에겐" / 우 "테토", 마커=scores.teto%. extro행 = 좌 "내향" / 우 "외향", 마커=scores.extro%. (구현자: renderResult/viz의 실제 DOM. 텍스트 textContent.)
